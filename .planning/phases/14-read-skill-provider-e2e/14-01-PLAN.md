```python
from __future__ import annotations
import logging, os
from pathlib import Path
from skills.discovery import discover_skills
from skills.frontmatter import parse_skill_doc
from skills.models import SkillInfo

logger = logging.getLogger(__name__)

class SkillStore:
    def __init__(self, root: Path) -> None:
        self._root = root
        self._case_insensitive = os.name == "nt"   # D-06: NTFS
        self._skills: dict[str, SkillInfo] = {}    # key = dedupe key
        self._reindex()

    def _key(self, name: str) -> str:
        return name.lower() if self._case_insensitive else name

    def _reindex(self) -> None:
        self._skills = {}
        for info in discover_skills(self._root):
            self._skills[self._key(info.name)] = info

    def lookup(self, name: str) -> SkillInfo:
        info = self._skills.get(self._key(name.strip()))
        if info is None:
            raise KeyError(f"Unknown skill '{name}'")   # caller formats the read_skill error
        return info

    def load(self, name: str) -> str:
        info = self.lookup(name)
        text = (info.path / "SKILL.md").read_text(encoding="utf-8")
        _meta, body = parse_skill_doc(text)
        return body.strip()

    def read_path(self, name: str, rel: str) -> str:
        info = self.lookup(name)
        base = info.path.resolve()                 # D-11: canonicalize base (resolves symlinks)
        p = (info.path / rel).resolve()            # D-11: canonicalize AFTER join — defeats ../ and symlink escapes
        if not p.is_relative_to(base):
            raise ValueError(f"Path '{rel}' escapes skill '{name}' — only files inside the skill directory are readable")  # D-12
        return p.read_text(encoding="utf-8")
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| LLM → `read_skill_path` arguments | Model-generated `path`/`skill` values are untrusted input crossing into the filesystem read path |
| Skill directory → context | Skill bodies and bundled resources are user-editable files — a malicious skill can instruct the agent to exfiltrate (prompt-injection) |
| Skill dir → host filesystem | `read_path` must never read outside the skill's own directory — the only sandbox in this phase |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-02 | Information Disclosure | `SkillStore.read_path` | mitigate | Canonicalize with `(skill_dir / rel).resolve()` then require `p.is_relative_to(skill_dir.resolve())`; reject non-relative input before resolution; D-12 error names skill + rejected path; win32 + posix vectors in the same wave (ASVS V9) |
| T-14-01 | Spoofing | `SkillStore.lookup` | mitigate | Frontmatter-name authority (D-04/D-05), case-insensitive match on win32 (D-06), no folder-name guessing |
| T-14-03 | Tampering | symlink/junction inside skill dir | mitigate | `Path.resolve()` canonicalization resolves symlinks before containment — a link resolving outside fails `is_relative_to` (D-11); escape test in suite |
</threat_model>
