    ```python
    from __future__ import annotations

    from collections import deque
    from itertools import cycle

    from textual.timer import Timer
    from textual.widgets import Input


    COMMANDS = ["/sessions", "/new", "/resume", "/title", "/help", "/exit"]

    PROCESSING_FRAMES = ["Thinking", "Thinking.", "Thinking..", "Thinking..."]
    PROCESSING_INTERVAL = 0.5


    class InputBar(Input):
        """Minimal single-line input bar with idle/processing visual states.

        Per D-10: Claude Code-style minimal input, no extra chrome.
        Per D-11: Visual state changes for idle vs processing (placeholder, color, disable).
        Per D-12: Preserves command history (up/down) and /command autocomplete.
        Per D-14: Processing state shown via placeholder animation and styling.

        Processing state applies a '--processing' CSS class (styled in theme.tcss).
        A cycling timer animates the placeholder during processing.
        """

        def __init__(self, **kwargs) -> None:
            super().__init__(placeholder="Type a prompt...", **kwargs)
            self._history: deque[str] = deque(maxlen=50)
            self._history_index: int = -1
            self._current_input: str = ""
            self._processing = False
            self._processing_cycle = cycle(PROCESSING_FRAMES)
            self._processing_timer: Timer | None = None

        @property
        def is_processing(self) -> bool:
            """Whether input is in processing/disabled state."""
            return self._processing

        def on_mount(self) -> None:
            """Create spinner timer (paused by default)."""
            self._processing_timer = self.set_interval(
                PROCESSING_INTERVAL, self._advance_processing, pause=True
            )

        def _advance_processing(self) -> None:
            """Cycle placeholder through Thinking... frames."""
            if self._processing_timer is not None:
                self.placeholder = next(self._processing_cycle)

        def update_processing(self, is_processing: bool) -> None:
            """Toggle processing state with placeholder animation and styling.

            Processing:
            - Disables input (prevents typing)
            - Sets 'Thinking' animated placeholder
            - Adds CSS class '--processing' for primary-color styling
            - Uses Textual animate() for smooth color transitions

            Idle:
            - Enables input
            - Restores placeholder
            - Removes '--processing' class
            """
            self._processing = is_processing
            if is_processing:
                self.disabled = True
                self.placeholder = "Thinking"
                self.add_class("--processing")
                self.styles.animate("color", value=self._processing_color, duration=0.3)
                if self._processing_timer:
                    self._processing_timer.resume()
            else:
                self.disabled = False
                self.placeholder = "Type a prompt..."
                self.remove_class("--processing")
                self.styles.animate("color", value=self._idle_color, duration=0.3)
                if self._processing_timer:
                    self._processing_timer.pause()

        @property
        def _idle_color(self) -> str:
            """Return the CSS color value for idle state."""
            return "$text"  # Resolved by Textual CSS variable

        @property
        def _processing_color(self) -> str:
            """Return the CSS color value for processing state."""
            return "$primary"  # Resolved by Textual CSS variable

        def on_key(self, event) -> None:
            if self.disabled:
                event.stop()
                return
            if event.key == "up":
                self._navigate_history(-1)
                event.stop()
            elif event.key == "down":
                self._navigate_history(1)
                event.stop()

        def _navigate_history(self, direction: int) -> None:
            if not self._history:
                return
            if self._history_index == -1:
                self._current_input = self.value
            new_index = self._history_index + direction
            if new_index < 0:
                self._history_index = -1
                self.value = self._current_input
            elif new_index >= len(self._history):
                self._history_index = len(self._history) - 1
            else:
                self._history_index = new_index
                self.value = list(self._history)[self._history_index]
            self.action_end()

        async def action_submit(self) -> None:
            value = self.value.strip()
            if value:
                self._history.append(value)
            self._history_index = -1
            self._current_input = ""
            await super().action_submit()
    ```
